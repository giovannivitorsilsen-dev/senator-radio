cmake_minimum_required(VERSION 3.16)
project(LifeServer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(life_resources resource_manager.cpp life_console.cpp life_events.cpp)
target_include_directories(life_resources PUBLIC .)
add_executable(life_resource_test test.cpp)
target_link_libraries(life_resource_test PRIVATE life_resources)
enable_testing()
add_test(NAME resource_test COMMAND life_resource_test)
add_executable(life_runtime_test runtime_test.cpp)
target_link_libraries(life_runtime_test PRIVATE life_resources)
add_test(NAME runtime_test COMMAND life_runtime_test)
add_executable(lifeserver-console life_console_main.cpp)
target_link_libraries(lifeserver-console PRIVATE life_resources)

option(LIFE_ENABLE_LUA "Compile embedded Lua 5.4 runtime" OFF)
if(LIFE_ENABLE_LUA)
  find_package(Lua 5.4 REQUIRED)
  add_library(life_lua lua_resource.cpp lua_host.cpp life_runtime.cpp life_runtime_console.cpp)
  target_include_directories(life_lua PRIVATE ${LUA_INCLUDE_DIR})
  target_link_libraries(life_lua PUBLIC life_resources ${LUA_LIBRARIES})
  target_include_directories(life_lua PUBLIC .)
  add_executable(life_lua_test lua_test.cpp)
  target_link_libraries(life_lua_test PRIVATE life_lua)
  add_test(NAME lua_runtime COMMAND life_lua_test)
  add_executable(life_integrated_test integrated_test.cpp)
  target_link_libraries(life_integrated_test PRIVATE life_lua)
  add_test(NAME integrated_runtime COMMAND life_integrated_test)
  add_executable(lifeserver-lua-console lua_console_main.cpp)
  target_link_libraries(lifeserver-lua-console PRIVATE life_lua)
  add_executable(lifeserver-lua-bridge lua_bridge_main.cpp)
  target_link_libraries(lifeserver-lua-bridge PRIVATE life_lua)
endif()
