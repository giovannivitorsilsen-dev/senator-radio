cmake_minimum_required(VERSION 3.16)
project(LifeServer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(life_resources resource_manager.cpp)
target_include_directories(life_resources PUBLIC .)
add_executable(life_resource_test test.cpp)
target_link_libraries(life_resource_test PRIVATE life_resources)
enable_testing()
add_test(NAME resource_test COMMAND life_resource_test)

option(LIFE_ENABLE_LUA "Compile embedded Lua 5.4 runtime" OFF)
if(LIFE_ENABLE_LUA)
  find_package(Lua 5.4 REQUIRED)
  add_library(life_lua lua_resource.cpp lua_host.cpp)
  target_include_directories(life_lua PRIVATE ${LUA_INCLUDE_DIR})
  target_link_libraries(life_lua PUBLIC life_resources ${LUA_LIBRARIES})
  target_include_directories(life_lua PUBLIC .)
  add_executable(life_lua_test lua_test.cpp)
  target_link_libraries(life_lua_test PRIVATE life_lua)
  add_test(NAME lua_runtime COMMAND life_lua_test)
endif()
